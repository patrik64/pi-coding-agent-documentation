Over its history the agent has changed where it keeps credentials, sessions, prompt templates, managed binaries and extensions. This file holds the startup code that moves old layouts to new ones so users never have to do it by hand. Each migration is idempotent: it checks whether there is anything to do, does it, and is a no-op afterwards.

`runMigrations` is the single entry point, called once at startup from [main.ts](/docs/src/main.ts). It returns the providers whose credentials were moved and a list of deprecation warnings; the caller decides how to show them, using `showDeprecationWarnings` for the blocking terminal prompt. The file depends on [config.ts](/docs/src/config.ts) for directory paths, on [keybindings.ts](/docs/src/core/keybindings.ts) for the keybinding config upgrade, and on [text.ts](/docs/src/utils/text.ts) for BOM stripping.

Migrations swallow almost every error. A failed migration should never stop the agent from starting, so most `catch` blocks only skip.

## !!steps Credentials into auth.json

`migrateAuthToAuthJson` is the oldest migration. It gathers OAuth entries from `oauth.json` and API keys from `settings.json` into one `auth.json`, the file now managed by [auth-storage.ts](/docs/src/core/auth-storage.ts). If `auth.json` already exists it does nothing at all.

The old `oauth.json` is renamed to `.migrated` rather than deleted, while the `apiKeys` entry is removed from settings and the file is rewritten. OAuth entries win over API keys for the same provider.

```ts ! migrations.ts
export function migrateAuthToAuthJson(): string[] {
	const agentDir = getAgentDir();
	const authPath = join(agentDir, "auth.json");
	const oauthPath = join(agentDir, "oauth.json");
	const settingsPath = join(agentDir, "settings.json");

	// Skip if auth.json already exists
	// !mark
	if (existsSync(authPath)) return [];

	const migrated: Record<string, unknown> = {};
	const providers: string[] = [];

	// Migrate oauth.json
	if (existsSync(oauthPath)) {
		try {
			const oauth = JSON.parse(stripBom(readFileSync(oauthPath, "utf-8")));
			for (const [provider, cred] of Object.entries(oauth)) {
				migrated[provider] = { type: "oauth", ...(cred as object) };
				providers.push(provider);
			}
			// !mark
			renameSync(oauthPath, `${oauthPath}.migrated`);
		} catch {
			// Skip on error
		}
	}
```

## !!steps Writing the result with private permissions

After both sources are read, the combined file is written with mode `0o600`, since it holds secrets. The function returns the provider names so startup can tell the user what moved.

```ts ! migrations.ts
	// Migrate settings.json apiKeys
	if (existsSync(settingsPath)) {
		try {
			const content = readFileSync(settingsPath, "utf-8");
			const settings = JSON.parse(stripBom(content));
			// !mark(1:3)
			if (settings.apiKeys && typeof settings.apiKeys === "object") {
				for (const [provider, key] of Object.entries(settings.apiKeys)) {
					if (!migrated[provider] && typeof key === "string") {
						migrated[provider] = { type: "api_key", key };
						providers.push(provider);
					}
				}
				// !mark(1:2)
				delete settings.apiKeys;
				writeFileSync(settingsPath, JSON.stringify(settings, null, 2));
			}
		} catch {
			// Skip on error
		}
	}

	if (Object.keys(migrated).length > 0) {
		mkdirSync(dirname(authPath), { recursive: true });
		// !mark
		writeFileSync(authPath, JSON.stringify(migrated, null, 2), { mode: 0o600 });
	}

	return providers;
}
```

## !!steps Rescuing sessions saved in the wrong place

A bug in an early version saved sessions as `.jsonl` files in the agent root instead of in a per-directory subfolder. This migration reads the header line of each such file, recomputes the directory name from the `cwd` (the same encoding as [session-manager.ts](/docs/src/core/session-manager.ts)), and moves the file. Existing targets are never overwritten, and files without a valid session header are left alone.

```ts ! migrations.ts
	for (const file of files) {
		try {
			// Read first line to get session header
			const content = readFileSync(file, "utf8");
			const firstLine = content.split("\n")[0];
			if (!firstLine?.trim()) continue;

			const header = JSON.parse(firstLine);
			// !mark
			if (header.type !== "session" || !header.cwd) continue;

			const cwd: string = header.cwd;

			// Compute the correct session directory (same encoding as session-manager.ts)
			// !mark
			const safePath = `--${cwd.replace(/^[/\\]/, "").replace(/[/\\:]/g, "-")}--`;
			const correctDir = join(agentDir, "sessions", safePath);

			// Create directory if needed
			if (!existsSync(correctDir)) {
				mkdirSync(correctDir, { recursive: true });
			}

			// Move the file
			const fileName = file.split("/").pop() || file.split("\\").pop();
			const newPath = join(correctDir, fileName!);

			// !mark
			if (existsSync(newPath)) continue; // Skip if target exists

			renameSync(file, newPath);
```

## !!steps Renaming commands to prompts

Prompt templates used to live in a `commands` directory. The rename only happens when `prompts` does not exist yet, and works for symlinks as well because it uses `renameSync`. A failure prints a warning and returns false instead of throwing.

```ts ! migrations.ts
function migrateCommandsToPrompts(baseDir: string, label: string): boolean {
	const commandsDir = join(baseDir, "commands");
	const promptsDir = join(baseDir, "prompts");

	// !mark
	if (existsSync(commandsDir) && !existsSync(promptsDir)) {
		try {
			// !mark(1:2)
			renameSync(commandsDir, promptsDir);
			console.log(chalk.green(`Migrated ${label} commands/ → prompts/`));
			return true;
		} catch (err) {
			console.log(
				chalk.yellow(
					`Warning: Could not migrate ${label} commands/ to prompts/: ${err instanceof Error ? err.message : err}`,
				),
			);
		}
	}
	return false;
}
```

## !!steps Upgrading keybindings and moving binaries

The keybindings file is parsed, passed through `migrateKeybindingsConfig`, and rewritten only if the function reports a change. Non-object JSON is ignored.

Managed `fd` and `rg` binaries moved from `tools/` to `bin/`. If the new path already has the binary, the old copy is deleted. The excerpt shows the keybindings part, then the binaries loop.

```ts ! migrations.ts
function migrateKeybindingsConfigFile(): void {
	const configPath = join(getAgentDir(), "keybindings.json");
	if (!existsSync(configPath)) return;

	try {
		const parsed = JSON.parse(stripBom(readFileSync(configPath, "utf-8"))) as unknown;
		if (typeof parsed !== "object" || parsed === null || Array.isArray(parsed)) {
			return;
		}
		// !mark(1:3)
		const { config, migrated } = migrateKeybindingsConfig(parsed as Record<string, unknown>);
		if (!migrated) return;
		writeFileSync(configPath, `${JSON.stringify(config, null, 2)}\n`, "utf-8");
	} catch {
		// Ignore malformed files during migration
	}
}
	// ...
	const binaries = ["fd", "rg", "fd.exe", "rg.exe"];
	let movedAny = false;

	for (const bin of binaries) {
		const oldPath = join(toolsDir, bin);
		const newPath = join(binDir, bin);

		if (existsSync(oldPath)) {
			if (!existsSync(binDir)) {
				mkdirSync(binDir, { recursive: true });
			}
			// !mark
			if (!existsSync(newPath)) {
				try {
					renameSync(oldPath, newPath);
					movedAny = true;
				} catch {
					// Ignore errors
				}
			// !mark(1:2)
			} else {
				// Target exists, just delete the old one
				try {
					rmSync?.(oldPath, { force: true });
				} catch {
					// Ignore
				}
			}
		}
	}
```

## !!steps Detecting deprecated extension directories

The old `hooks/` and `tools/` directories no longer load as extensions. This function only produces warnings; it moves nothing. Because `tools/` also holds extracted `fd` and `rg` binaries, it only warns when the directory contains something else, ignoring hidden files such as `.DS_Store`.

```ts ! migrations.ts
function checkDeprecatedExtensionDirs(baseDir: string, label: string): string[] {
	const hooksDir = join(baseDir, "hooks");
	const toolsDir = join(baseDir, "tools");
	const warnings: string[] = [];

	// !mark
	if (existsSync(hooksDir)) {
		warnings.push(`${label} hooks/ directory found. Hooks have been renamed to extensions.`);
	}

	if (existsSync(toolsDir)) {
		// Check if tools/ contains anything other than fd/rg (which are auto-extracted binaries)
		try {
			const entries = readdirSync(toolsDir);
			// !mark(1:6)
			const customTools = entries.filter((e) => {
				const lower = e.toLowerCase();
				return (
					lower !== "fd" && lower !== "rg" && lower !== "fd.exe" && lower !== "rg.exe" && !e.startsWith(".") // Ignore .DS_Store and other hidden files
				);
			});
			if (customTools.length > 0) {
				warnings.push(
					`${label} tools/ directory contains custom tools. Custom tools have been merged into extensions.`,
				);
			}
		} catch {
			// Ignore read errors
		}
	}

	return warnings;
}
```

## !!steps Collecting warnings and prompting

`migrateExtensionSystem` runs the commands rename for both the global and the project directory, then collects warnings for both. `showDeprecationWarnings` prints them in yellow with links to the migration guide and then waits for a single keypress in raw mode, so the warnings are not scrolled away by the interface that starts next.

```ts ! migrations.ts
function migrateExtensionSystem(cwd: string): string[] {
	const agentDir = getAgentDir();
	const projectDir = join(cwd, CONFIG_DIR_NAME);

	// Migrate commands/ to prompts/
	// !mark(1:2)
	migrateCommandsToPrompts(agentDir, "Global");
	migrateCommandsToPrompts(projectDir, "Project");

	// Check for deprecated directories
	const warnings = [
		...checkDeprecatedExtensionDirs(agentDir, "Global"),
		...checkDeprecatedExtensionDirs(projectDir, "Project"),
	];

	return warnings;
}
// ...
export async function showDeprecationWarnings(warnings: string[]): Promise<void> {
	if (warnings.length === 0) return;

	for (const warning of warnings) {
		console.log(chalk.yellow(`Warning: ${warning}`));
	}
	console.log(chalk.yellow(`\nMove your extensions to the extensions/ directory.`));
	console.log(chalk.yellow(`Migration guide: ${MIGRATION_GUIDE_URL}`));
	console.log(chalk.yellow(`Documentation: ${EXTENSIONS_DOC_URL}`));
	console.log(chalk.dim(`\nPress any key to continue...`));

	await new Promise<void>((resolve) => {
		// !mark
		process.stdin.setRawMode?.(true);
		process.stdin.resume();
		// !mark
		process.stdin.once("data", () => {
			process.stdin.setRawMode?.(false);
			process.stdin.pause();
			resolve();
		});
	});
	console.log();
}
```

## !!steps The entry point and its order

`runMigrations` runs the migrations in a fixed order: credentials, misplaced sessions, binaries, keybindings, and finally the extension system. The extension migration is last since it is the only one that depends on the project directory and the only one that produces user-facing warnings.

```ts ! migrations.ts
export function runMigrations(cwd: string): {
	migratedAuthProviders: string[];
	deprecationWarnings: string[];
} {
	// !mark(1:5)
	const migratedAuthProviders = migrateAuthToAuthJson();
	migrateSessionsFromAgentRoot();
	migrateToolsToBin();
	migrateKeybindingsConfigFile();
	const deprecationWarnings = migrateExtensionSystem(cwd);
	return { migratedAuthProviders, deprecationWarnings };
}
```

</Scrollycoding>

Everything here is synchronous except the keypress prompt, and only `migrateAuthToAuthJson`, `migrateSessionsFromAgentRoot`, `showDeprecationWarnings` and `runMigrations` are exported. `runMigrations` is the one place to read or extend the sequence.
